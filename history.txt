Filter: Must be a file<\n>        if ($file | path type) != "file" { return 0 }<\n><\n>        # 2. Filter: Check exclusions against path components<\n>        let components = ($file | path split)<\n>        if ($components | any {|part| $part in $exclude}) { return 0 }<\n><\n>        # 3. Execution: Count lines safely<\n>        # We use 'open --raw' to avoid auto-parsing formats like JSON/TOML,<\n>        # treating everything as text. 'try' handles binary read failures.<\n>        try {<\n>            open --raw $file | lines | length<\n>        } catch {<\n>            # Returns 0 for binaries or permission-locked files<\n>            0 <\n>        }<\n>    }<\n>    | math sum<\n>}
